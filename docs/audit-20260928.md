## Evidencia de esquema y permisos

Consultas ejecutadas:

```sql
SELECT schemaname,tablename,policyname,roles::text,cmd,qual,with_check
FROM pg_policies
WHERE schemaname IN ('public','storage')
  AND tablename IN ('members','group_socials','products','product_variants',
    'product_images','profiles','posts','post_images','events','objects','buckets')
ORDER BY schemaname,tablename,policyname;

SELECT n.nspname||'.'||c.relname AS relation,c.relrowsecurity,c.relforcerowsecurity,
       coalesce(string_agg(p.policyname||':'||p.cmd||':'||p.roles::text,
         ', ' ORDER BY p.policyname),'(none)') policies
FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace
LEFT JOIN pg_policies p ON p.schemaname=n.nspname AND p.tablename=c.relname
WHERE (n.nspname='public' AND c.relname IN
  ('members','group_socials','products','product_variants','product_images',
   'profiles','posts','post_images','events'))
  OR (n.nspname='storage' AND c.relname IN ('objects','buckets'))
GROUP BY n.nspname,c.relname,c.relrowsecurity,c.relforcerowsecurity ORDER BY 1;

SELECT id,public,file_size_limit,allowed_mime_types FROM storage.buckets
WHERE id IN ('gimae-blog','gimae-products') ORDER BY id;
```
